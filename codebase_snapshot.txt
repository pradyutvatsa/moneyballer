import 'package:flutter/material.dart';
import 'package:amplify_authenticator/amplify_authenticator.dart';
import 'package:amplify_flutter/amplify_flutter.dart';
import 'package:amplify_api/amplify_api.dart';
import 'package:moneyballer/models/ModelProvider.dart';
import 'package:moneyballer/features/scheduling/create_game_screen.dart';

class DashboardScreen extends StatefulWidget {
  const DashboardScreen({super.key});

  @override
  State<DashboardScreen> createState() => _DashboardScreenState();
}

class _DashboardScreenState extends State<DashboardScreen> {
  List<Game> _games = [];
  bool _isLoading = true;

  @override
  void initState() {
    super.initState();
    _fetchGames();
  }

  Future<void> _fetchGames() async {
    try {
      if (!mounted) return;
      setState(() => _isLoading = true);

      final request = ModelQueries.list(Game.classType);
      final response = await Amplify.API.query(request: request).response;

      if (response.hasErrors) {
        safePrint('GraphQL Query errors: ${response.errors}');
      }

      final items = response.data?.items;
      safePrint('Raw items received: ${items?.length}');

      if (items != null) {
        final fetchedGames = items.whereType<Game>().toList();
        safePrint('Parsed ${fetchedGames.length} valid games!');

        if (mounted) {
          setState(() {
            _games = List<Game>.from(fetchedGames);
            _isLoading = false;
          });
        }
      } else {
        if (mounted) {
          setState(() {
            _games = [];
            _isLoading = false;
          });
        }
      }
    } on ApiException catch (e) {
      safePrint('Query failed: $e');
      if (mounted) setState(() => _isLoading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.grey[200],
      appBar: AppBar(
        title: const Text(
          'MoneyBaller',
          style: TextStyle(fontSize: 24, fontWeight: FontWeight.w900),
        ),
        backgroundColor: Colors.black,
        foregroundColor: Colors.white,
        actions: const [SignOutButton()],
      ),
      body: _isLoading
          ? const Center(child: CircularProgressIndicator(color: Colors.black))
          : RefreshIndicator(
              onRefresh: _fetchGames,
              color: Colors.black,
              child: ListView(
                padding: const EdgeInsets.all(16.0),
                children: [
                  const Text(
                    'Upcoming Games',
                    style: TextStyle(
                      fontSize: 32,
                      fontWeight: FontWeight.w900,
                      color: Colors.black,
                    ),
                  ),
                  const SizedBox(height: 24),
                  if (_games.isEmpty)
                    const Padding(
                      padding: EdgeInsets.only(top: 40.0),
                      child: Center(
                        child: Text(
                          'No games scheduled yet.',
                          style: TextStyle(fontSize: 20, color: Colors.grey),
                        ),
                      ),
                    ),
                  for (final game in _games) ...[
                    Padding(
                      padding: const EdgeInsets.only(bottom: 16.0),
                      child: _buildEventCard(
                        title: 'Game (${game.maxPlayers} Max)',
                        date:
                            '${game.date.format()} at ${game.time.format().substring(0, 5)}',
                        location: game.location,
                      ),
                    ),
                  ],
                ],
              ),
            ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: () async {
          await Navigator.push<bool>(
            context,
            MaterialPageRoute(builder: (context) => const CreateGameScreen()),
          );
          await _fetchGames();
        },
        backgroundColor: Colors.black,
        foregroundColor: Colors.white,
        icon: const Icon(Icons.add, size: 28),
        label: const Text(
          'Create Game',
          style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
        ),
      ),
    );
  }

  Widget _buildEventCard({
    required String title,
    required String date,
    required String location,
  }) {
    return Card(
      elevation: 6,
      color: Colors.white,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
      child: Padding(
        padding: const EdgeInsets.all(20.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              title,
              style: const TextStyle(
                fontSize: 26,
                fontWeight: FontWeight.w900,
                color: Colors.black,
              ),
            ),
            const SizedBox(height: 12),
            Row(
              children: [
                const Icon(
                  Icons.calendar_today,
                  size: 24,
                  color: Colors.black87,
                ),
                const SizedBox(width: 12),
                Text(
                  date,
                  style: const TextStyle(
                    fontSize: 20,
                    color: Colors.black87,
                    fontWeight: FontWeight.w600,
                  ),
                ),
              ],
            ),
            const SizedBox(height: 12),
            Row(
              children: [
                const Icon(Icons.location_on, size: 24, color: Colors.black87),
                const SizedBox(width: 12),
                Text(
                  location,
                  style: const TextStyle(
                    fontSize: 20,
                    color: Colors.black87,
                    fontWeight: FontWeight.w600,
                  ),
                ),
              ],
            ),
          ],
        ),

    );
  }

              fontWeight: FontWeight.w900,
              color: Colors.black,
            ),
          ),
          const SizedBox(height: 24),

          _buildInfoRow(Icons.calendar_today, game.date.format()),
          const SizedBox(height: 16),

          _buildInfoRow(Icons.access_time, game.time.format().substring(0, 5)),
          const SizedBox(height: 16),

          _buildInfoRow(Icons.location_on, game.location),
          const SizedBox(height: 40),

          const Text(
            'RSVP Status',
            style: TextStyle(
              fontSize: 24,
              fontWeight: FontWeight.bold,
              color: Colors.black,
            ),
          ),
          const SizedBox(height: 16),

          // RSVP Action Buttons
          Row(
            children: [
              Expanded(
                child: ElevatedButton(
                  style: ElevatedButton.styleFrom(
                    backgroundColor: Colors.black,
                    foregroundColor: Colors.white,
                  ),
                  onPressed: () async {
                    final double playerShare = game.maxPlayers > 0
                        ? (game.appFee) / game.maxPlayers
                        : 0.0;

                    try {
                      final authUser = await Amplify.Auth.getCurrentUser();
                      final userId = authUser.userId;

                      final playerRequest = ModelQueries.list(
                        User.classType,
                        where: User.ID.eq(userId),
                      );
                      final playerResponse = await Amplify.API
                          .query(request: playerRequest)
                          .response;
                      final players = playerResponse.data?.items;

                      if (players == null || players.isEmpty) return;
                      final player = players.first;
                      if (player == null) return;

                      if (player.walletBalance < playerShare) {
                        if (context.mounted) {
                          ScaffoldMessenger.of(context).showSnackBar(
                            SnackBar(
                              content: Text(
                                'Insufficient funds. Top up \$$playerShare',
                              ),
                              backgroundColor: Colors.red,
                            ),
                          );
                        }
                        return;
                      }

                      final updatedPlayer = player.copyWith(
                        walletBalance: player.walletBalance - playerShare,
                      );
                      await Amplify.API
                          .mutate(request: ModelMutations.update(updatedPlayer))
                          .response;

                      final orgRequest = ModelQueries.list(
                        User.classType,
                        where: User.ID.eq(game.organizerId),
                      );
                      final orgResponse = await Amplify.API
                          .query(request: orgRequest)
                          .response;
                      final organizers = orgResponse.data?.items;

                      if (organizers != null &&
                          organizers.isNotEmpty &&
                          organizers.first != null) {
                        final organizer = organizers.first!;
                        final updatedOrganizer = organizer.copyWith(
                          walletBalance: organizer.walletBalance + playerShare,
                        );
                        await Amplify.API
                            .mutate(
                              request: ModelMutations.update(updatedOrganizer),
                            )
                            .response;
                      }

                      final playerPayment = Payment(
                        amount: playerShare,
                        type: 'RSVP Fee Debit',
                        status: 'Completed',
                        user: updatedPlayer,
                      );
                      await Amplify.API
                          .mutate(request: ModelMutations.create(playerPayment))
                          .response;

                      final newRsvp = RSVP(
                        status: 'Going',
                        game: game,
                        user: updatedPlayer,
                      );
                      final rsvpResponse = await Amplify.API
                          .mutate(request: ModelMutations.create(newRsvp))
                          .response;

                      if (!rsvpResponse.hasErrors && context.mounted) {
                        ScaffoldMessenger.of(context).showSnackBar(
                          const SnackBar(
                            content: Text('You are marked as GOING!'),
                            backgroundColor: Colors.green,
                          ),
                        );
                      }
                    } on ApiException catch (e) {
                      safePrint('RSVP Transaction failed: $e');
                    }
                  },
                  child: const Padding(
                    padding: EdgeInsets.symmetric(vertical: 20),
                    child: Text(
                      'GOING',
                      style: TextStyle(
                        fontSize: 20,
                        fontWeight: FontWeight.w900,
                      ),
                    ),
                  ),
                ),
              ),
              const SizedBox(width: 16),
              Expanded(
                child: ElevatedButton(
                  style: ElevatedButton.styleFrom(
                    backgroundColor: Colors.white,
                    foregroundColor: Colors.black,
                    side: const BorderSide(color: Colors.black, width: 2),
                  ),
                  onPressed: () async {
                    try {
                      final authUser = await Amplify.Auth.getCurrentUser();
                      final userId = authUser.userId;

                      final userRequest = ModelQueries.list(
                        User.classType,
                        where: User.ID.eq(userId),
                      );
                      final userResponse = await Amplify.API
                          .query(request: userRequest)
                          .response;
                      final users = userResponse.data?.items;

                      if (users == null ||
                          users.isEmpty ||
                          users.first == null) {
                        return;
                      }
                      final player = users.first!;

                      final newRsvp = RSVP(
                        status: 'Not Going',
                        game: game,
                        user: player,
                      );

                      final request = ModelMutations.create(newRsvp);
                      final response = await Amplify.API
                          .mutate(request: request)
                          .response;

                      if (response.hasErrors) {
                        safePrint('RSVP Errors: ${response.errors}');
                      } else {
                        safePrint(

                        if (context.mounted) {

                              backgroundColor: Colors.orange,

                      }
                    } on ApiException catch (e) {

// Ensure these imports are at the top of create_game_screen.dart:
import 'package:flutter/material.dart';
import 'package:amplify_flutter/amplify_flutter.dart';
import 'package:amplify_api/amplify_api.dart';
import 'package:moneyballer/models/ModelProvider.dart';

class CreateGameScreen extends StatefulWidget {
  const CreateGameScreen({super.key});

  @override
  State<CreateGameScreen> createState() => _CreateGameScreenState();
}

class _CreateGameScreenState extends State<CreateGameScreen> {
  final _formKey = GlobalKey<FormState>();
  final _locationController = TextEditingController();
  final _maxPlayersController = TextEditingController();

  DateTime? _selectedDate;
  TimeOfDay? _selectedTime;

  Future<void> _pickDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: DateTime.now(),
      firstDate: DateTime.now(),
      lastDate: DateTime(2100),
      builder: (context, child) => Theme(
        data: ThemeData.light().copyWith(
          colorScheme: const ColorScheme.light(primary: Colors.black),
        ),
        child: child!,
      ),
    );
    if (picked != null) setState(() => _selectedDate = picked);
  }

  Future<void> _pickTime() async {
    final picked = await showTimePicker(
      context: context,
      initialTime: TimeOfDay.now(),
      builder: (context, child) => Theme(
        data: ThemeData.light().copyWith(
          colorScheme: const ColorScheme.light(primary: Colors.black),
        ),
        child: child!,
      ),
    );
    if (picked != null) setState(() => _selectedTime = picked);
  }

  @override
  void dispose() {
    _locationController.dispose();
    _maxPlayersController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.grey[200],
      appBar: AppBar(
        title: const Text(
          'Create Game',
          style: TextStyle(fontWeight: FontWeight.w900),
        ),
        backgroundColor: Colors.black,
        foregroundColor: Colors.white,
      ),
      body: Form(
        key: _formKey,
        child: ListView(
          padding: const EdgeInsets.all(16.0),
          children: [
            _buildLabel('Date'),
            _buildPickerField(
              text: _selectedDate == null
                  ? 'Select Date'
                  : '${_selectedDate!.toLocal()}'.split(' ')[0],
              icon: Icons.calendar_today,
              onTap: _pickDate,
            ),
            const SizedBox(height: 24),

            _buildLabel('Time'),
            _buildPickerField(
              text: _selectedTime == null
                  ? 'Select Time'
                  : _selectedTime!.format(context),
              icon: Icons.access_time,
              onTap: _pickTime,
            ),
            const SizedBox(height: 24),

            _buildLabel('Location'),
            TextFormField(
              controller: _locationController,
              style: const TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
              decoration: _inputStyle('Enter field or park name'),
              validator: (value) => value == null || value.isEmpty
                  ? 'Location is required'
                  : null,
            ),
            const SizedBox(height: 24),

            _buildLabel('Max Players limit'),
            TextFormField(
              controller: _maxPlayersController,
              keyboardType: TextInputType.number,
              style: const TextStyle(fontSize: 20, fontWeight: FontWeight.bold),
              decoration: _inputStyle('e.g., 10 or 20'),
              validator: (value) =>
                  value == null || value.isEmpty ? 'Limit is required' : null,
            ),
            const SizedBox(height: 40),

            ElevatedButton(
              style: ElevatedButton.styleFrom(
                backgroundColor: Colors.black,
                foregroundColor: Colors.white,
                padding: const EdgeInsets.symmetric(vertical: 20),
                shape: RoundedRectangleBorder(
                  borderRadius: BorderRadius.circular(12),
                ),
              ),
              onPressed: () async {
                if (_formKey.currentState!.validate() &&
                    _selectedDate != null &&
                    _selectedTime != null) {
                  final int maxPlayers = int.parse(_maxPlayersController.text);
                  // Calculate the upfront organizer float fee based on session size[cite: 1]
                  final double fee = maxPlayers <= 10 ? 1.99 : 2.99;

                  try {
                    // 1. Fetch the authenticated Cognito User
                    final authUser = await Amplify.Auth.getCurrentUser();
                    final userId = authUser.userId;

                    // 2. Fetch the Organizer's current wallet balance from DynamoDB
                    final userRequest = ModelQueries.list(
                      User.classType,
                      where: User.ID.eq(userId),
                    );
                    final userResponse = await Amplify.API
                        .query(request: userRequest)
                        .response;
                    final users = userResponse.data?.items;

                    if (users == null || users.isEmpty || users.first == null) {
                      safePrint('Organizer profile not found in DynamoDB.');
                      return;
                    }
                    final organizer = users.first!;

                    // 3. Verify sufficient funds
                    if (organizer.walletBalance < fee) {
                      if (context.mounted) {
                        ScaffoldMessenger.of(context).showSnackBar(
                          SnackBar(
                            content: Text(
                              'Insufficient funds. Please top up \$$fee to schedule.',
                            ),
                            backgroundColor: Colors.red,
                          ),
                        );
                      }
                      return;
                    }

                    // 4. Deduct the upfront fee from the organizer's wallet[cite: 1]
                    final updatedOrganizer = organizer.copyWith(
                      walletBalance: organizer.walletBalance - fee,
                    );
                    await Amplify.API
                        .mutate(
                          request: ModelMutations.update(updatedOrganizer),
                        )
                        .response;

                    // 5. Generate the Payment receipt using the full user object[cite: 5]
                    final payment = Payment(
                      amount: fee,
                      type: 'Organizer Fee Debit',
                      status: 'Completed',
                      user: organizer,
                    );
                    await Amplify.API
                        .mutate(request: ModelMutations.create(payment))
                        .response;

                    // 6. Schedule the Game[cite: 1]
                    final combinedDateTime = DateTime(
                      _selectedDate!.year,
                      _selectedDate!.month,
                      _selectedDate!.day,
                      _selectedTime!.hour,
                      _selectedTime!.minute,
                    );

                    final newGame = Game(
                      date: TemporalDate(_selectedDate!),
                      time: TemporalTime(combinedDateTime),
                      location: _locationController.text,
                      organizerId: userId,
                      maxPlayers: maxPlayers,
                      appFee: fee,
                      minimumBalanceRequired: fee,
                    );

                    final gameResponse = await Amplify.API
                        .mutate(request: ModelMutations.create(newGame))
                        .response;

                    if (gameResponse.hasErrors) {
                      safePrint('Game creation errors: ${gameResponse.errors}');
                    } else {
                      safePrint(
                        'Game scheduled and \$$fee deducted successfully!',
                      );
                      if (context.mounted) Navigator.pop(context, true);
                    }
                  } on ApiException catch (e) {
                    safePrint('Transaction failed: $e');
                  }
                }
              },
              child: const Text(
                'SCHEDULE GAME',
                style: TextStyle(fontSize: 24, fontWeight: FontWeight.w900),
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildLabel(String text) {
    return Padding(
      padding: const EdgeInsets.only(bottom: 8.0),
      child: Text(
        text,
        style: const TextStyle(
          fontSize: 22,
          fontWeight: FontWeight.w900,
          color: Colors.black,
        ),
      ),
    );
  }

  InputDecoration _inputStyle(String hint) {
    return InputDecoration(
      hintText: hint,
      filled: true,
      fillColor: Colors.white,
      border: OutlineInputBorder(
        borderRadius: BorderRadius.circular(8),
        borderSide: const BorderSide(color: Colors.grey),
      ),
      focusedBorder: OutlineInputBorder(
        borderRadius: BorderRadius.circular(8),
        borderSide: const BorderSide(color: Colors.black, width: 3),
      ),
    );
  }

  Widget _buildPickerField({
    required String text,
    required IconData icon,
    required VoidCallback onTap,
  }) {
    return InkWell(
      onTap: onTap,
      child: Container(
        padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 20),
        decoration: BoxDecoration(
          color: Colors.white,
          border: Border.all(color: Colors.grey),
          borderRadius: BorderRadius.circular(8),
        ),
        child: Row(
          children: [
            Icon(icon, size: 28, color: Colors.black87),
            const SizedBox(width: 16),
            Text(
              text,
              style: const TextStyle(
                fontSize: 20,
                fontWeight: FontWeight.bold,
                color: Colors.black,
              ),
            ),
          ],
        ),
      ),
    );
  }
}

import type { PostConfirmationTriggerHandler } from 'aws-lambda';
import { DynamoDBClient } from '@aws-sdk/client-dynamodb';
import { DynamoDBDocumentClient, PutCommand } from '@aws-sdk/lib-dynamodb';

const client = new DynamoDBClient();
const docClient = DynamoDBDocumentClient.from(client);

export const handler: PostConfirmationTriggerHandler = async (event) => {
    if (event.request.userAttributes) {
        const { sub, email, name } = event.request.userAttributes;

        // Create the new User record based on your Gen 2 schema[cite: 1]
        const command = new PutCommand({
            TableName: process.env.USER_TABLE_NAME, // Injected via backend.ts
            Item: {
                id: sub,
                __typename: 'User',
                email: email,
                name: name || 'New Player',
                walletBalance: 0.0, // Initialize empty wallet for RSVPs[cite: 1]
                createdAt: new Date().toISOString(),
                updatedAt: new Date().toISOString(),
            },
        });

        try {
            await docClient.send(command);
            console.log(`Successfully created DynamoDB User record for ${sub}`);
        } catch (error) {
            console.error(`Error syncing user to DynamoDB: ${error}`);
        }
    }

    return event;
};
